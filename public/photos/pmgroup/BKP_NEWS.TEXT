    <script>
        document.addEventListener("DOMContentLoaded", function () {
            // Collapsible functionality
            const collapsibles = document.getElementsByClassName("collapsible");
            for (let i = 0; i < collapsibles.length; i++) {
                collapsibles[i].addEventListener("click", function() {
                    this.classList.toggle("active");
                    const content = this.nextElementSibling;
                    if (content.style.maxHeight) {
                        content.style.maxHeight = null;
                    } else {
                        content.style.maxHeight = content.scrollHeight + "px";
                    }
                });
            }
            
            // JavaScript fallback for seamless scrolling if needed
            const newsList = document.getElementById('newsList');
            const newsContainer = document.querySelector('.news-container');
            
            // Duplicate items for JavaScript-based seamless scrolling
            function duplicateItemsForSeamlessScroll() {
                const items = newsList.querySelectorAll('li');
                const containerHeight = newsContainer.offsetHeight;
                let totalHeight = 0;
                
                // Calculate total height of original items
                items.forEach(item => {
                    totalHeight += item.offsetHeight;
                });
                
                // If the total height is less than container height, duplicate until it fills
                if (totalHeight < containerHeight) {
                    const originalItems = Array.from(items);
                    const clonesNeeded = Math.ceil(containerHeight / totalHeight);
                    
                    for (let i = 0; i < clonesNeeded; i++) {
                        originalItems.forEach(item => {
                            const clone = item.cloneNode(true);
                            newsList.appendChild(clone);
                        });
                    }
                }
            }
            
            // Call the function to ensure seamless scrolling
            duplicateItemsForSeamlessScroll();
            
            // Reset animation when it completes for true seamless effect
            newsList.addEventListener('animationiteration', function() {
                // This ensures the animation restarts seamlessly
            });
        });
    </script>
